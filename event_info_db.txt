CREATE TABLE IF NOT EXISTS stoxscoop_dev.event_batches
(
    id integer NOT NULL DEFAULT nextval('stoxscoop.event_batches_id_seq'::regclass),
    batch_name character varying(255) COLLATE pg_catalog."default" NOT NULL,
    ingestion_source ingestion_source_enum NOT NULL DEFAULT 'manual'::ingestion_source_enum,
    notes text COLLATE pg_catalog."default",
    started_at timestamp with time zone NOT NULL DEFAULT now(),
    completed_at timestamp with time zone,
    total_events integer DEFAULT 0,
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    updated_at timestamp with time zone,
    CONSTRAINT event_batches_pkey PRIMARY KEY (id)
)

CREATE TABLE IF NOT EXISTS stoxscoop_dev.events
(
    id integer NOT NULL DEFAULT nextval('stoxscoop.events_id_seq'::regclass),
    stock_id integer NOT NULL,
    batch_id integer,
    event_type event_type_enum NOT NULL,
    event_subtype character varying(80) COLLATE pg_catalog."default" NOT NULL,
    signal_type signal_type_enum NOT NULL DEFAULT 'neutral'::signal_type_enum,
    signal_reason text COLLATE pg_catalog."default",
    sentiment sentiment_enum NOT NULL DEFAULT 'neutral'::sentiment_enum,
    priority priority_enum NOT NULL DEFAULT 'low'::priority_enum,
    impact_score integer,
    confidence_score numeric(4,3),
    confidence_model_version character varying(30) COLLATE pg_catalog."default",
    tags text[] COLLATE pg_catalog."default" DEFAULT '{}'::text[],
    title character varying(500) COLLATE pg_catalog."default" NOT NULL,
    summary text COLLATE pg_catalog."default",
    event_date date NOT NULL,
    source_url text COLLATE pg_catalog."default",
    source_name character varying(100) COLLATE pg_catalog."default",
    ingestion_source ingestion_source_enum NOT NULL DEFAULT 'manual'::ingestion_source_enum,
    is_verified boolean NOT NULL DEFAULT false,
    is_active boolean NOT NULL DEFAULT true,
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    updated_at timestamp with time zone,
    CONSTRAINT events_pkey PRIMARY KEY (id),
    CONSTRAINT events_batch_id_fkey FOREIGN KEY (batch_id)
        REFERENCES stoxscoop_dev.event_batches (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE SET NULL,
    CONSTRAINT fk_events_stock FOREIGN KEY (stock_id)
        REFERENCES classification.ticker_symbol (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT fk_events_subtype FOREIGN KEY (event_type, event_subtype)
        REFERENCES stoxscoop_dev.event_subtypes (event_type, subtype_code) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT chk_confidence_version CHECK (confidence_score IS NULL OR confidence_model_version IS NOT NULL)
)
